if args.sample and items:
    print("\n[Field Values]")

    check_fields = [
        "productCode",
        "pname",
        "eventName",
        "eventSourceType",
        "act",
        "actResult",
        "firstAct",
        "firstActResult",
        "aggregatedCount",
        "reason",
        "msg",
        "remarks"
    ]

    for field in check_fields:
        values = {
            json.dumps(
                item[field],
                ensure_ascii=False,
                sort_keys=True
            )
            for item in items
            if field in item
        }

        print(f"\n[{field}]")

        if values:
            for value in sorted(values):
                print(f" - {value}")
        else:
            print(" - Field not found")
